% Clean ISOPE caption formatting (flush baseline, zero overfull hbox)
\long\def\@makecaption#1#2{%
  \vskip 4\p@
  \footnotesize\baselineskip=9.8pt
  \sbox\@tempboxa{#1 #2}%
  \ifdim \wd\@tempboxa >\hsize
    \noindent #1 #2\par
  \else
    \global \@minipagefalse
    \hb@xt@\hsize{\hfil\box\@tempboxa\hfil}%
  \fi
  \vskip 2\p@
}
